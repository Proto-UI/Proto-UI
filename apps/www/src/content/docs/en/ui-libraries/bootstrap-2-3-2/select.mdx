## Installation

Currently available from workspace source. Import all eleven parts from `@proto.ui/prototypes-bootstrap-2-3-2/select`:

```ts
import {
  selectRoot,
  selectTrigger,
  selectValue,
  selectContent,
  selectSurface,
  selectViewport,
  selectItem,
  selectItemText,
  selectGroup,
  selectLabel,
  selectSeparator,
} from '@proto.ui/prototypes-bootstrap-2-3-2/select';
```

## Usage

```tsx
import {
  Bootstrap232SelectContent,
  Bootstrap232SelectGroup,
  Bootstrap232SelectItem,
  Bootstrap232SelectItemText,
  Bootstrap232SelectLabel,
  Bootstrap232SelectRoot,
  Bootstrap232SelectSurface,
  Bootstrap232SelectTrigger,
  Bootstrap232SelectValue,
  Bootstrap232SelectViewport,
} from '../proto-ui/components/react';

export function FruitSelect() {
  return (
    <Bootstrap232SelectRoot defaultValue="apple">
      <Bootstrap232SelectTrigger className="w-48" accessibleLabel="Fruit selector">
        <Bootstrap232SelectValue placeholder="Select a fruit" />
      </Bootstrap232SelectTrigger>
      <Bootstrap232SelectContent>
        <Bootstrap232SelectSurface>
          <Bootstrap232SelectViewport>
            <Bootstrap232SelectGroup>
              <Bootstrap232SelectLabel>Fruits</Bootstrap232SelectLabel>
              <Bootstrap232SelectItem value="apple" textValue="Apple">
                <Bootstrap232SelectItemText>Apple</Bootstrap232SelectItemText>
              </Bootstrap232SelectItem>
              <Bootstrap232SelectItem value="banana" textValue="Banana">
                <Bootstrap232SelectItemText>Banana</Bootstrap232SelectItemText>
              </Bootstrap232SelectItem>
            </Bootstrap232SelectGroup>
          </Bootstrap232SelectViewport>
        </Bootstrap232SelectSurface>
      </Bootstrap232SelectContent>
    </Bootstrap232SelectRoot>
  );
}
```

### Composition

```text
SelectRoot
├── SelectTrigger
│   └── SelectValue
└── SelectContent
    └── SelectSurface
        └── SelectViewport
            ├── SelectGroup
            │   ├── SelectLabel
            │   └── SelectItem
            │       └── SelectItemText
            └── SelectSeparator
```

- **Classic Bootstrap 2.3.2 styling**: Trigger pairs `.btn` gradients with an 8×4 caret; the popup reproduces `.dropdown-menu` 6px corners, drop shadow, 11px uppercase `.nav-header` labels, two-tone `.divider` separators, and blue gradient active rows.
- **Keyboard & accessibility**: Supports `Enter`, `Space`, arrow navigation, typeahead, and `Escape` dismissal with focus restoration. Long labels wrap cleanly and follow logical text direction.
